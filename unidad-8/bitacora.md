# Actividad 02 
En el momento en el que una estrella explota, no desaparece: transforma todo lo que la compone. Los planetas, la luna, los recuerdos… todo se dispersa para formar algo nuevo. De la misma manera, la mente humana no crea desde la nada, sino que utiliza fragmentos de lo vivido, lo recordado y lo sentido para construir nuevas realidades. Una supernova no puede existir sin un sol previo, y precisamente son los restos de ese sol los que se expanden para dar origen a nuevas formas.

Este es el concepto que quiero representar en mi obra: la transformación a partir de lo que ya existe, la creación que surge del colapso y la memoria.

Me inspiro en la obra Sónar+D CCCB 2020 Music: "Carles Viarnès & Alba G. Corral 360º AV Show", en la cual las visuales varían en tamaño y forma según la velocidad y la intensidad de la música. ## Modo Sistema Solar / Supernova (Nuevo)
Este modo marca el punto medio de la canción, cuando la composición entra en su momento más calmado y, al mismo tiempo, en el único compás tradicional (4/4). Aquí se representa un sistema solar que rota lentamente, estable y equilibrado, como si todo hubiera encontrado finalmente su orden.

Sin embargo, esta calma es solo aparente. Mientras la canción avanza, comienza a acumular energía, construyendo poco a poco hacia la explosión: la supernova. En ese instante, todo cambia.

La música rompe todas las normas rítmicas que había seguido hasta ese punto —abandona la estabilidad del 4/4 y vuelve a compases irregulares como 11/8, 9/4 y 10/7—, y de la misma forma, en la simulación todas las reglas anteriores se quiebran.

Pero así como una supernova está compuesta de lo que una vez fue su estrella, esta nueva etapa no destruye completamente lo anterior, sino que lo transforma. Las gotas, los planetas, las raíces y los renacuajos siguen existiendo, pero ya no como eran antes:

En el paisaje comienzan a caer gotas en perspectivas imposibles.

Gotas y planetas aparecen simultáneamente, sin sentido espacial.

En el lago de renacuajos comienzan a crecer raíces mientras nuevos sistemas solares se forman a gran velocidad.

Todo converge en la última explosión, la última supernova, donde cada fragmento de los modos anteriores se funde en una misma expansión.

const RIPPLE_LIFE = 1000;

const PERC_MARGIN = 170;

const PALETTE = [
  [255,  99,  71],[255, 184,  77],[ 80, 220, 130],

let snd, fft, peak;
let audioReady = false, audioPlaying = false;
const PEAK_FREQ_LOW = 20, PEAK_FREQ_HIGH = 180, PEAK_THRESHOLD = 0.22;

class ModoLluvia {
  constructor() {
    this.gotas = [];
    this.ondas = [];
    this.bg = color(...BG_LLUVIA);
    fft = new p5.FFT(0.9, 1024);
    peak = new p5.PeakDetect(PEAK_FREQ_LOW, PEAK_FREQ_HIGH, PEAK_THRESHOLD, 20);
  }
  onEnter(){ background(this.bg); }
  onResize(){ background(this.bg); }
  draw() {
    background(this.bg);
    if (audioPlaying) {
      fft.analyze();
      peak.update(fft);
      this.detectPercusion();
    }
    for (let g of this.gotas) { g.update(); g.display(); }
    for (let i = this.gotas.length - 1; i >= 0; i--) {
      if (this.gotas[i].isDead()) {
        this.gotas[i].impact(this.ondas);
        this.gotas.splice(i, 1);
      }
    }
    for (let o of this.ondas) { o.update(); o.display(); }
    this.ondas = this.ondas.filter(o => !o.isDead());
  }
  agregarGotaNormal() {
    const x = random(width  * 0.05, width  * 0.95);
    const y = random(height * 0.10, height * 0.95);
    this.gotas.push(new GotaNormal(x, y));
  }
  forzarImpactoPrimero() {
    let idx = -1, oldest = Infinity;
    for (let i = 0; i < this.gotas.length; i++) {
      if (this.gotas[i].birth < oldest) { oldest = this.gotas[i].birth; idx = i; }
    }
    if (idx >= 0) {
      this.gotas[idx].impact(this.ondas);
      this.gotas.splice(idx, 1);
    }
  }
  detectPercusion() {
    const now = millis();
    if (peak.isDetected && now - lastPercAt > PEAK_COOLDOWN_MS) {
      lastPercAt = now;
      const pos = this.randomCorner(PERC_MARGIN);
      this.gotas.push(new GotaPercusion(pos.x, pos.y));
    }
  }
  randomCorner(margin = 0) {
    const m = min(margin, width * 0.25, height * 0.25);
    const i = floor(random(4));
    return [
      createVector(m, m),
      createVector(width - m, m),
      createVector(width - m, height - m),
      createVector(m, height - m),
    ][i];
  }
  startMusic() {
    userStartAudio().then(() => {
      if (!audioReady) { fft.setInput(snd); audioReady = true; }
      if (!snd.isPlaying()) { snd.loop(); audioPlaying = true; }
    }).catch(()=>{});
  }
}

class GotaBase {
  constructor(x, yFinal) {
    this.pos = createVector(x, yFinal);
    this.yFinal = yFinal;
    this.birth = millis();
    this.lifespan = PARTICLE_LIFESPAN;
    this.rMax = random(70, 110);
    this.rMin = random(6, 10);
    const c = random(PALETTE);
    this.col = color(c[0], c[1], c[2], 220);
  }
  progress(){ return constrain((millis() - this.birth) / this.lifespan, 0, 1); }
  currentRadius(){ return lerp(this.rMax, this.rMin, this.progress()); }
  update(){}
  display(){}
  impact(ondasArr){}
  isDead(){ return millis() - this.birth >= this.lifespan; }
}

class OndaBase {
  constructor(x, y, col) {
    this.pos = createVector(x, y);
    this.birth = millis();
    this.lifespan = RIPPLE_LIFE;
    this.maxR = random(90, 140);
    this.col = col || color(255);
  }
  t(){ return constrain((millis() - this.birth) / this.lifespan, 0, 1); }
  isDead(){ return millis() - this.birth >= this.lifespan; }
}
class OndaCircular extends OndaBase {
  update(){}
  display() {
    const k = this.t();
    const r = this.maxR * (1 - pow(1 - k, 3));
    const a = map(1 - k, 0, 1, 0, 230);
    noFill(); stroke(red(this.col), green(this.col), blue(this.col), a); strokeWeight(3);
    circle(this.pos.x, this.pos.y, r * 2);
  }
}
class OndaPoligonal extends OndaBase {
  constructor(x, y, sides, col){ super(x, y, col); this.sides = sides; }
  update(){}
  display() {
    const k = this.t();
    const r = this.maxR * (1 - pow(1 - k, 3));
    const a = map(1 - k, 0, 1, 0, 230);
    noFill(); stroke(red(this.col), green(this.col), blue(this.col), a); strokeWeight(4);
    push(); translate(this.pos.x, this.pos.y); beginShape();
    for (let i = 0; i < this.sides; i++) {
      const ang = TWO_PI * (i / this.sides);
      vertex(r * cos(ang), r * sin(ang));
    }
    endShape(CLOSE); pop();
  }
}

class Sun {
  constructor(tx, ty, m, c) {
    this.tx = tx; this.ty = ty;
    this.mass = m; this.r = sqrt(this.mass) * 0.7;
    this.c = c;
    this.anchor = createVector(random(-width*0.35, width*0.35),
                               random(-height*0.35, height*0.35));
    this.scale = random(120, 240);
    this.speed = random(0.002, 0.0045);
    this.pos = this.anchor.copy();
  }
  update() {
    const nx = noise(this.tx) * 2 - 1;
    const ny = noise(this.ty) * 2 - 1;
    this.pos.set(this.anchor.x + nx * this.scale,
                 this.anchor.y + ny * this.scale);
    this.tx += this.speed;
    this.ty += this.speed * 0.97;
  }
  show() {
    push(); translate(this.pos.x, this.pos.y); noStroke();
    for (let i = 6; i > 0; i--) {
      fill(red(this.c), green(this.c), blue(this.c), 30);
      ellipse(0, 0, this.r * i * 0.7);
    }
    fill(this.c);
    beginShape();
    let spikes = 10, step = TWO_PI / spikes;
    for (let a = 0; a < TWO_PI; a += step) {
      let ro = this.r * 1.4, ri = this.r * 0.5;
      vertex(cos(a) * ro, sin(a) * ro);
      vertex(cos(a + step/2) * ri, sin(a + step/2) * ri);
    }
    endShape(CLOSE); pop();
  }
}
class Planet {
  constructor(parent, orbitRadius, size, speed) {
    this.parent = parent;
    this.orbitRadius = orbitRadius;
    this.size = size;
    this.speed = speed;
    this.angle = random(TWO_PI);
    this.col = color(random(255), random(255), random(255));
  }
  update(){ this.angle += this.speed; }
  show() {

class ModoPeces {
  constructor(){
    this.sepDist = 25; this.aliDist = 50; this.cohDist = 50;
    this.wrapEnabled = true;
    this.bgChoices = [
      [12,54,130],[10,69,150],[14,84,170],[8,60,140]
    ];
    this.bg = color(...this.bgChoices[0]);
    this.flock = new Flock();
    for (let i = 0; i < 300; i++)
      this.flock.addBoid(new BoidP(random(width), random(height)));
    this.grouping = false;
    this.scatterUntil = 0;
  }
  onEnter(){ background(this.bg); }
  onResize(){ background(this.bg); }
  draw(){
    background(this.bg);
    // fuerzas de modo (extra) sobre boids
    const center = createVector(width/2, height/2);
    for (let b of this.flock.boids) {
      b.wrap = this.wrapEnabled;
      b.setNeighborhood(this.sepDist, this.aliDist, this.cohDist);
      // agrupación suave
      if (this.grouping) {
        const toC = p5.Vector.sub(center, b.position).setMag(0.15);
        b.applyForce(toC);
      }
      // dispersión breve (impulso)
      if (millis() < this.scatterUntil) {
        let away = p5.Vector.random2D().mult(0.3);
        b.applyForce(away);
      }
      b.run(this.flock.boids);
    }
  }
  agruparCentro(){ this.grouping = true; }
  dispersar(){
    this.grouping = false;
    this.scatterUntil = millis() + 1100; // ~1.1s de “huida”
    // dar un “empujón” inicial a cada pez
    for (let b of this.flock.boids) {
      const dir = p5.Vector.random2D().mult(2.2);
      b.velocity.add(dir).limit(b.maxspeed);
    }
  }
}

class BoidP {
  constructor(x, y) {
    this.acceleration = createVector(0, 0);
    this.velocity = p5.Vector.random2D().mult(random(1.2,2.2));
    this.position = createVector(x, y);
    this.r = 4;
    this.maxspeed = 4.2;
    this.maxforce = 0.07;
    this.sepDist=25; this.aliDist=50; this.cohDist=50;
    const c = random(PALETTE); this.fillCol = color(c[0],c[1],c[2]);
    this.wrap = true;
  }
  setNeighborhood(s,a,c){ this.sepDist=s; this.aliDist=a; this.cohDist=c; }
  applyForce(f){ this.acceleration.add(f); }
  run(boids){
    const sep = this.separate(boids).mult(1.4);
    const ali = this.align(boids).mult(1.0);
    const coh = this.cohere(boids).mult(1.0);
    this.applyForce(sep); this.applyForce(ali); this.applyForce(coh);
    this.update(); this.borders(); this.show();
  }
  update(){
    this.velocity.add(this.acceleration);
    this.velocity.limit(this.maxspeed);
    this.position.add(this.velocity);
    this.acceleration.mult(0);
  }
  borders(){
    if (this.wrap) {
      if (this.position.x < -this.r) this.position.x = width + this.r;
      if (this.position.y < -this.r) this.position.y = height + this.r;
      if (this.position.x > width + this.r) this.position.x = -this.r;
      if (this.position.y > height + this.r) this.position.y = -this.r;
    } else {
      if (this.position.x < this.r) { this.position.x=this.r; this.velocity.x*=-1; }
      if (this.position.x > width-this.r) { this.position.x=width-this.r; this.velocity.x*=-1; }
      if (this.position.y < this.r) { this.position.y=this.r; this.velocity.y*=-1; }
      if (this.position.y > height-this.r) { this.position.y=height-this.r; this.velocity.y*=-1; }
    }
  }
  seek(target){
    let desired = p5.Vector.sub(target, this.position).setMag(this.maxspeed);
    let steer = p5.Vector.sub(desired, this.velocity);
    steer.limit(this.maxforce);
    return steer;
  }
  show(){
    const ang = this.velocity.heading();
    push();
    translate(this.position.x, this.position.y);
    rotate(ang);
    noStroke();
    fill(this.fillCol);
    ellipse(0, 0, 12, 8);
    fill(255, 180);
    ellipse(-6, 0, 8, 5);
    pop();
  }
  separate(boids){
    let steer = createVector(0,0), count=0;
    for (let o of boids){
      let d = p5.Vector.dist(this.position, o.position);
      if (d > 0 && d < this.sepDist){
        let diff = p5.Vector.sub(this.position, o.position).normalize().div(d);
        steer.add(diff); count++;
      }
    }
    if (count>0) steer.div(count);
    if (steer.mag()>0) steer.normalize().mult(this.maxspeed).sub(this.velocity).limit(this.maxforce);
    return steer;
  }
  align(boids){
    let sum = createVector(0,0), count=0;
    for (let o of boids){
      let d = p5.Vector.dist(this.position, o.position);
      if (d>0 && d < this.aliDist){ sum.add(o.velocity); count++; }
    }
    if (count>0){ sum.div(count).normalize().mult(this.maxspeed);
      let steer = p5.Vector.sub(sum, this.velocity).limit(this.maxforce); return steer; }
    return createVector(0,0);
  }
  cohere(boids){
    let sum = createVector(0,0), count=0;
    for (let o of boids){
      let d = p5.Vector.dist(this.position, o.position);
      if (d>0 && d<this.cohDist){ sum.add(o.position); count++; }
    }
    if (count>0){ sum.div(count); return this.seek(sum); }
    return createVector(0,0);
  }
}

class ModoArbol {
  constructor() {
    this.center = createVector(width/2, height/2);
    this.branches = [];
    this.maxBranches = 4000;    
    this.spawnPerFrame = 16;   
    this.running = false;      
    this.collapsing = false;   
  }

  onEnter() {
    background(10, 10, 20);
    this.resetAndStart();      
  }

  onResize() {
    this.center.set(width/2, height/2);
    if (!this.running && !this.collapsing) background(10,10,20);
  }


  grow() { this.resetAndStart(); }


  shrink() {
    if (!this.branches.length) return;
    this.running = false;
    this.collapsing = true;
  }

  resetAndStart() {
    this.branches = [
      new Rama(this.center.x, this.center.y, random(TWO_PI), this.randomCol(), 0)
    ];
    this.running = true;
    this.collapsing = false;
    background(10, 10, 20);
  }

  draw() {

    background(10, 10, 20, 40);


    if (this.collapsing) {
   
      let allCollapsed = true;
      for (let b of this.branches) {
        b.collapse(this.center); 
        b.display();
        if (!b.isCollapsed(this.center)) allCollapsed = false;
      }

      if (allCollapsed) {
        this.collapsing = false;
        this.running = false;
        this.branches = [ new Rama(this.center.x, this.center.y, random(TWO_PI), this.randomCol(), 0) ];

      }
      return; 
    }


    for (let b of this.branches) {
      b.update();
      b.display();
    }

    if (!this.running) return;

 
    let spawned = 0;
    if (this.branches.length < this.maxBranches) {
      for (let tries = 0; tries < this.spawnPerFrame && spawned < this.spawnPerFrame; tries++) {
        const parent = random(this.branches);
        if (!parent || !parent.end || !parent.active) continue;

        const numChildren = random() < 0.65 ? 2 : 1;
        for (let k = 0; k < numChildren; k++) {
          if (this.branches.length >= this.maxBranches) break;
          const bend = random(-PI/3, PI/3);
          this.branches.push(
            new Rama(parent.end.x, parent.end.y, parent.angle + bend, this.randomCol(), parent.depth + 1)
          );
          spawned++;
          if (spawned >= this.spawnPerFrame) break;
        }
      }
    } else {
      this.running = false; 
    }
  }

  randomCol() {

    return color(random(100,255), random(100,255), random(100,255));
  }
}

class Rama {
  constructor(x, y, angle, col, depth) {
    this.start = createVector(x, y);
    this.angle = angle;
    this.col = col;
    this.depth = depth || 0;

    const baseMax = random(500, 300);
    this.maxLen = max(40, baseMax * (1.0 - 0.08 * this.depth)); 
    this.len = 0;
    this.growStep = random(5, 12);
    this.active = true;
    this.end = null;
    this._collapsed = false;
  }

  update() {
    if (this.active && this.len < this.maxLen) {
      const jitter = sin((frameCount + this.start.x + this.start.y) * 0.02) * 0.25;
      this.len += this.growStep + jitter;
      if (this.len >= this.maxLen) { this.len = this.maxLen; this.active = false; }
    } else {
      this.active = false;
    }
    this.end = p5.Vector.add(this.start, p5.Vector.fromAngle(this.angle).mult(this.len));
  }

  // Animación de colapso hacia el centro
  collapse(center) {
    if (!this.end) {
      this.end = p5.Vector.add(this.start, p5.Vector.fromAngle(this.angle).mult(this.len));
    }
    const k = 0.12; // factor de compresión por frame
    this.start.lerp(center, k);
    this.end.lerp(center, k);
    // actualizar longitud según la nueva geometría
    this.len = p5.Vector.dist(this.start, this.end);
    // marcar cuando ya está esencialmente en el centro
    if (this.len < 1.2 && p5.Vector.dist(this.end, center) < 1.5) this._collapsed = true;
  }

  isCollapsed(center) {
    return this._collapsed === true;
  }

  display() {
    const w = map(this.len, 0, this.maxLen, 5, 0.8);
    stroke(this.col);
    strokeWeight(w);
    line(this.start.x, this.start.y, this.end.x, this.end.y);
  }
}

/* ===================== MODO 0: SUPERNOVA (Z) ===================== */
class ModoSupernova0 {
  constructor() {
    this.seedPalette = ["#4DA6FF", "#FFFFFF", "#B0BEC5", "#2E7D32", "#9C27B0", "#FFC107", "#E91E63"];
    this.palette = [];
    this.sunColor = "#FFD700";
    this.sunChoices = ["#FFD700", "#ff4242", "#6aa4ff", "#ffffff"];
    this.PALETTE_BIAS = 0.70;

    this.sunCoreR = 120;
    this.sunHaloR = 400;
    this.planets = [];

    this.phase = "normal"; // normal | supernova
    this.sparks = [];
    this.novaStart = 0;
    this.NOVA_DURATION = 1200;
    this.baseOmega = 0.0011; // velocidad orbital base
  }

  onEnter() {
    this.initSystem(true);
    background(5,10,20);
  }
  onResize() {
    // reajusta escalas cuando cambia el canvas
    const oldCore = this.sunCoreR || 1;
    this.sunCoreR = min(width, height) * 0.08;
    this.sunHaloR = this.sunCoreR * 3.4;
    const k = this.sunCoreR / oldCore;
    for (const p of this.planets) { p.ae *= k; p.b *= k; }
  }

  draw() {
    background(5, 10, 20, this.phase === "supernova" ? 140 : 70);
    push();
    translate(width/2, height/2);
    if (this.phase === "supernova") {
      this.updateSupernova();
    } else {
      this.drawSun(false);
      for (const p of this.planets) { p.update(); p.show(); }
    }
    pop();
  }

  // Acción única del modo: Z → supernova
  trigger() { this.triggerSupernova(); }

  /* --------- lógica del sistema --------- */
  initSystem(first=false) {
    const sunScale = random(0.8, 1.2);
    this.sunCoreR = min(width, height) * 0.08 * sunScale;
    this.sunHaloR = this.sunCoreR * 3.4;

    this.sunColor = random(this.sunChoices);

    const base = (first || this.palette.length === 0)
      ? this.seedPalette.slice()
      : this.palette.map(c => c.toString('#rrggbb'));
    this.palette = [];
    for (let i = 0; i < 7; i++) this.palette.push(this.chooseBiasedColor(base));

    this.planets = [];
    const margin = 40 * sunScale;
    const aMin = this.sunHaloR + margin;
    const step = min(width, height) * 0.075 * sunScale;

    for (let i = 0; i < 7; i++) {
      const a = aMin + i * step;
      const e = random(0.05, 0.55);
      const b = a * sqrt(1 - e * e);
      const omega = this.baseOmega * pow(aMin / a, 1.5) * random(0.85, 1.15);
      const argPeri = random(TWO_PI);
      const col = this.palette[i];
      const styles = ['blob','oblate','ringed','striped','spiky','diamond','amoeba','twins','comet','gasGiant','lavaWorld','crystal','ringedNebula','shattered'];
      const style = random(styles);
      const size = map(i, 0, 6, 14, 28) * random(0.8, 1.35) * sunScale;

      this.planets.push(new SuperPlanetEllipse(a, b, omega, random(TWO_PI), argPeri, col, size, style));
    }

    this.phase = "normal";
    this.sparks.length = 0;
  }

  chooseBiasedColor(baseHexes) {
    if (random() > this.PALETTE_BIAS) return color(random(255), random(255), random(255));
    const base = color(random(baseHexes));
    colorMode(HSB, 360, 100, 100, 255);
    const h = (hue(base) + random(-22, 22) + 360) % 360;
    const s = constrain(saturation(base) + random(-18, 18), 15, 100);
    const b = constrain(brightness(base) + random(-18, 18), 18, 100);
    const c = color(h, s, b, 255);
    colorMode(RGB, 255);
    return c;
  }

  triggerSupernova() {
    if (this.phase === "supernova") return;
    this.phase = "supernova";
    this.novaStart = frameCount;
    this.sparks = [];
    const N = 2500;
    for (let i = 0; i < N; i++) {
      const ang = random(TWO_PI);
      const spd = random(8, 25);
      const vel = p5.Vector.fromAngle(ang).mult(spd);
      const col = (random() < 0.6) ? color(this.sunColor) : random(this.palette);
      this.sparks.push({
        pos: createVector(0, 0),
        vel,
        life: random(90, 200),
        c: col,
        size: random(2, 8)
      });
    }
  }

  updateSupernova() {
    const t = (frameCount - this.novaStart);
    const tNorm = constrain(t / this.NOVA_DURATION, 0, 1);
    const maxDim = max(width, height);

    push();
    noStroke();

    // flash
    const flashAlpha = map(t, 0, 120, 255, 0, true);
    fill(255, 240, 180, flashAlpha);
    rect(-width/2, -height/2, width, height);

    // shockwave
    const shockwaveR = map(t, 0, 300, 0, maxDim * 1.6, true);
    noFill();
    stroke(255, 220, 150, map(t, 0, 240, 180, 0, true));
    strokeWeight(map(t, 0, 120, 12, 2, true));
    circle(0, 0, shockwaveR);
    pop();

    // fondo pulsante
    colorMode(HSB, 360, 100, 100, 255);
    const hueShift = (t * 0.5) % 360;
    const bgAlpha = map(sin(t * 0.02), -1, 1, 10, 40) * (1 - tNorm);
    fill((hue(color(this.sunColor)) + hueShift) % 360, 80, 70, bgAlpha);
    noStroke();
    rect(-width/2, -height/2, width, height);
    colorMode(RGB, 255);

    // chispas
    for (const s of this.sparks) {
      s.pos.add(s.vel);
      s.vel.mult(0.985);
      s.life--;
      noStroke();
      const a = map(s.life, 0, 200, 0, 255, true);
      fill(red(s.c), green(s.c), blue(s.c), a);
      circle(s.pos.x, s.pos.y, s.size);
    }
    this.sparks = this.sparks.filter(s => s.life > 0);

    // desvanecer a negro justo antes del reset
    if (t > this.NOVA_DURATION - 180) {
      const fadeAlpha = map(t, this.NOVA_DURATION - 180, this.NOVA_DURATION, 0, 255, true);
      fill(0, 0, 0, fadeAlpha);
      rect(-width/2, -height/2, width, height);
    }

    // reiniciar sistema
    if (t > this.NOVA_DURATION || this.sparks.length === 0) {
      this.initSystem(false);
    }
  }

  drawSun(isNova) {
    noStroke();
    const layers = isNova ? 14 : 10;
    for (let i = layers; i >= 1; i--) {
      const alpha = isNova ? (18 + i * 8) : (16 + i * 6);
      const sc = color(this.sunColor);
      fill(red(sc), green(sc), blue(sc), alpha);
      const r = map(i, 1, layers, this.sunCoreR * 1.8, this.sunHaloR * 2.1);
      circle(0, 0, r);
    }
    fill(this.sunColor);
    const core = isNova ? this.sunCoreR * 1.2 : this.sunCoreR;
    circle(0, 0, core * 2);
    // brillo sutil (sin UI ni textos)
    fill(255, 255, 255, isNova ? 110 : 85);
    ellipse(core * 0.22, -core * 0.22, core * 0.5, core * 0.35);
  }
}


class SuperPlanetEllipse {
  constructor(a, b, omega, angle, argPeri, col, size, style) {
    this.ae = a; this.b = b; this.omega = omega;
    this.a = angle; this.arg = argPeri;
    this.c = (typeof col === 'string') ? color(col) : col;
    this.size = size; this.style = style;
    this.spin = random(-0.02, 0.02);
    this.wobble = random(0.015, 0.05);
    this.seed = random(10000);
  }
  update(){ this.a += this.omega; }
  show() {
    const ct = cos(this.a), st = sin(this.a);
    const x0 = this.ae * ct, y0 = this.b * st;
    const ca = cos(this.arg), sa = sin(this.arg);
    const x = ca * x0 - sa * y0;
    const y = sa * x0 + ca * y0;
    push(); translate(x, y); rotate(frameCount * this.spin);

    // halo
    noStroke();
    for (let i = 4; i >= 1; i--) {
      const alpha = 10 * i;
      fill(red(this.c), green(this.c), blue(this.c), alpha);
      circle(0, 0, this.size * (1 + i * 0.25));
    }


    const c = this.c;
    if (this.style === 'oblate') {
      push(); rotate(-PI/9);
      fill(c);
      const wob = 1 + 0.12 * sin(frameCount * this.wobble);
      ellipse(0,0, this.size*1.45*wob, this.size*0.85/wob);
      pop();
    } else if (this.style === 'ringed') {
      fill(c); circle(0,0,this.size);
      noFill(); stroke(230,230,230,160); strokeWeight(2);
      push(); rotate(-PI/7 + 0.2 * sin(frameCount * this.wobble));
      ellipse(0,0,this.size*1.9,this.size*0.65);
      stroke(255,255,255,90);
      ellipse(0,0,this.size*2.2,this.size*0.75);
      pop();
    } else if (this.style === 'striped') {
      fill(c); circle(0,0,this.size);
      noFill(); stroke(255,255,255,40); strokeWeight(1.25);
      arc(0,0, this.size*0.95, this.size*0.62, -PI*0.2, PI*0.8);
      arc(0,0, this.size*0.75, this.size*0.48, -PI*0.2, PI*0.8);
    } else if (this.style === 'spiky') {
      fill(c); beginShape();
      let pts = int(random(8,14));
      for (let i=0;i<pts;i++){
        let ang = (TWO_PI/pts)*i;
        let r = this.size*0.45*(1.2 + 0.6*noise(this.seed + i*0.3 + frameCount*0.01));
        vertex(cos(ang)*r*2, sin(ang)*r*2);
      }
      endShape(CLOSE);
    } else if (this.style === 'diamond') {
      fill(c);
      const s = this.size*0.85;
      beginShape();
      vertex(0,-s); vertex(s*0.9,0); vertex(0,s); vertex(-s*0.9,0);
      endShape(CLOSE);
    } else if (this.style === 'amoeba') {
      fill(c); beginShape();
      const steps = 24;

        const ls = this.size*0.3 + this.size*0.2*sin(frameCount*0.03 + i);

/* ===================== MODO 5: PAISAJE ANIMADO ===================== */
class ModoPaisaje5 {
  constructor(){
    this.t = 0;

    // gotas y ondas (reutiliza Modo 1)
    this.gotas = [];
    this.ondas = [];

    // colores base
    this.skyTop  = color(12,18,32);
    this.skyBot  = color(28,44,72);
    this.sunCol  = color(255,200,60);

    // capas de paisaje
    this.hills = [];     // {yBase, amp, speed, col, seed}
    this.grass = [];     // {y, amp, speed, alpha}
    this.clouds = [];    // {x,y,s,seed,speed,alpha}

    // detalles extra
    this.trees = [];     // {x, yOff, s, layer, hueJit}
    this.flocks = [];    // aves: {x, y, vx, amp, members, scale, alpha}

    this.buildScenery();
  }

  buildScenery(){
    this.hills.length = 0;
    this.grass.length = 0;
    this.clouds.length = 0;
    this.trees.length = 0;
    this.flocks.length = 0;

    // Montañas (de atrás hacia adelante)
    const layers = 5;
    for (let i=0;i<layers;i++){
      const yBase = map(i, 0, layers-1, height*0.35, height*0.72);
      const amp   = map(i, 0, layers-1, 60, 18);
      const spd   = map(i, 0, layers-1, 0.04, 0.18);
      const c     = color(40+ i*20, 60+ i*18, 90+ i*12, 170 - i*20);
      this.hills.push({yBase: yBase, amp: amp, speed: spd, col: c, seed: random(10000)});
    }

    // Pradera
    for (let i=0;i<8;i++){
      const y = map(i, 0, 7, height*0.7, height*0.98);
      this.grass.push({y, amp: random(4,12), speed: random(0.8,1.4), alpha: map(i,0,7,40,110)});
    }

    // Nubes
    for (let i=0;i<12;i++){
      this.clouds.push({
        x: random(-width, width*1.2),
        y: random(height*0.06, height*0.35),
        s: random(60, 200),
        seed: random(10000),
        speed: random(0.12, 0.28),
        alpha: random(45, 90)
      });
    }

    // Árboles (3 planos con tamaños/colores variados)
    const bands = [
      {count: 28, layer: 0, sMin: 0.5, sMax: 0.9, yOff: -26},
      {count: 34, layer: 2, sMin: 0.7, sMax: 1.2, yOff: -18},
      {count: 38, layer: 4, sMin: 1.0, sMax: 1.7, yOff:  -8},
    ];
    for (const b of bands){
      for (let i=0;i<b.count;i++){
        this.trees.push({
          x: random(width),
          yOff: b.yOff + random(-8,8),
          s: random(b.sMin, b.sMax),
          layer: b.layer,
          hueJit: random(-10, 14)
        });
      }
    }

    // Bandadas de aves
    const flocksN = 3;
    for (let i=0;i<flocksN;i++){
      this.flocks.push({
        x: random(-width*0.2, width*1.2),
        y: random(height*0.10, height*0.28),
        vx: random(0.8, 1.8) * (random()<0.5 ? 1 : -1),
        amp: random(6, 14),
        members: int(random(5, 9)),
        scale: random(0.5, 0.9),
        alpha: random(70, 120)
      });
    }
  }

  onEnter(){ background(12,18,32); }
  onResize(){ this.buildScenery(); background(12,18,32); }

  // ---- Interacción ----
  agregarGota(){     // Z: gota en cualquier lugar (fix aleatorio)
    const x = random(width*0.03,  width*0.97);
    const y = random(height*0.06, height*0.97);
    this.gotas.push(new GotaNormal(x, y));
  }
  forzarImpacto(){   // X
    let idx=-1, oldest=Infinity;
    for (let i=0;i<this.gotas.length;i++){
      if (this.gotas[i].birth < oldest){ oldest=this.gotas[i].birth; idx=i; }
    }
    if (idx>=0){ this.gotas[idx].impact(this.ondas); this.gotas.splice(idx,1); }
  }

  draw(){
    this.t += 0.01;

    // Cielo en gradiente
    const breathe = (sin(frameCount*0.002)+1)/2;
    const topCol = lerpColor(this.skyTop, color(10,16,28), breathe*0.25);
    const botCol = lerpColor(this.skyBot, color(22,38,68), breathe*0.25);
    for (let y=0; y<height; y+=2){
      const k = y/height;
      stroke(lerpColor(topCol, botCol, k));
      line(0,y,width,y);
    }

    // Sol con halo
    push();
    translate(width*0.5, height*0.28);
    noStroke();
    for (let i=14;i>=1;i--){
      fill(255, 200, 64, 10 + i*7);
      circle(0,0, i*26);
    }
    fill(this.sunCol);
    circle(0,0, 150);
    pop();

    // Nubes (sin tocar el generador global de random)
    for (let cl of this.clouds){
      cl.x += cl.speed;
      if (cl.x > width + cl.s*1.2) cl.x = -cl.s*1.2;
      if (cl.x < -cl.s*1.2)       cl.x = width + cl.s*1.2;

      push();
      translate(cl.x, cl.y);
      noStroke();
      fill(240, 245, 255, cl.alpha);
      this.blobCloudLocalRNG(cl.s, cl.seed);
      pop();
    }

    // Montañas
    for (let i=0;i<this.hills.length;i++){
      const h = this.hills[i];
      this.drawHillLayer(h);
    }

    // Árboles (en el suelo del layer elegido)
    for (const tr of this.trees){
      const ground = this.groundY(tr.x, this.hills[tr.layer]);
      this.drawTree(tr.x, ground + tr.yOff, tr.s, tr.hueJit);
    }

    // Pradera
    for (let g of this.grass){
      noStroke();
      fill(34, 120, 70, g.alpha);
      const wav = sin(this.t * g.speed + g.y*0.02) * g.amp;
      rect(0, g.y + wav, width, height - g.y - wav);
    }

    // Aves de fondo
    for (let f of this.flocks){
      f.x += f.vx;
      if (f.x > width + 100)  f.x = -100;
      if (f.x < -100)         f.x = width + 100;

      this.drawFlock(f);
    }

    // Gotas y ondas
    for (let g of this.gotas){ g.update(); g.display(); }
    for (let i=this.gotas.length-1;i>=0;i--){
      if (this.gotas[i].isDead()){
        this.gotas[i].impact(this.ondas);
        this.gotas.splice(i,1);
      }
    }
    for (let o of this.ondas){ o.update(); o.display(); }
    this.ondas = this.ondas.filter(o=>!o.isDead());
  }


  _rng(seed){ // Mulberry32
    let t = seed >>> 0;
    return () => {
      t += 0x6D2B79F5;
      let r = Math.imul(t ^ t >>> 15, 1 | t);
      r ^= r + Math.imul(r ^ r >>> 7, 61 | r);
      return ((r ^ r >>> 14) >>> 0) / 4294967296;
    };
  }

  blobCloudLocalRNG(S, seed){
    const rnd = this._rng((seed*1000)|0);
    const n = 10;
    for (let i=0;i<n;i++){
      const r = lerp(S*0.28, S*0.5, rnd());
      const a = lerp(-PI/6, PI/6, rnd());
      const ox = cos(a)*S*0.15 + lerp(-S*0.12, S*0.12, rnd());
      const oy = sin(a)*S*0.08 + lerp(-S*0.06, S*0.06, rnd());
      ellipse(ox, oy, r*2, r*1.5);
    }
  }

  drawHillLayer(h){
    noStroke();
    fill(h.col);
    beginShape();
    vertex(0, height);
    for (let x=0; x<=width; x+=8){
      const nx = (x*0.002) + this.t*h.speed + h.seed;
      const y = h.yBase + (noise(nx)*2-1) * h.amp;
      vertex(x, y);
    }
    vertex(width, height);
    endShape(CLOSE);

    // luz suave en cresta
    stroke(255,255,255, 18);
    noFill();
    beginShape();
    for (let x=0; x<=width; x+=8){
      const nx = (x*0.002) + this.t*h.speed + h.seed + 100;
      const y = h.yBase + (noise(nx)*2-1) * (h.amp*0.75);
      vertex(x, y-2);
    }
    endShape();
  }

  groundY(x, hill){
    const nx = (x*0.002) + this.t*hill.speed + hill.seed;
    return hill.yBase + (noise(nx)*2-1) * hill.amp;
  }

  drawTree(x, y, s, hueJit){
    push();
    translate(x, y);
    scale(s);

    // tronco
    noStroke();
    fill(115, 70, 38, 220);
    rect(-3, 0, 6, 18, 2);

    // copa (tres triángulos)
    const g = color(28+hueJit, 120+hueJit*0.4, 60+hueJit*0.2, 230);
    fill(g);
    triangle(0, -34, -16, -8, 16, -8);
    triangle(0, -48, -14, -22, 14, -22);
    triangle(0, -60, -12, -36, 12, -36);
    pop();
  }

  drawFlock(f){
    push();
    translate(f.x, f.y);
    scale(f.scale, f.scale);
    stroke(240, 230, 210, f.alpha);
    strokeWeight(1.6);
    noFill();

    for (let i=0;i<f.members;i++){
      const px = i*14;
      const py = sin((frameCount*0.08 + i)*0.7) * f.amp * 0.25;
  
      beginShape();
      vertex(px-6, py);
      vertex(px,   py+3);
      vertex(px+6, py);
      endShape();
    }
    pop();
  }
}
